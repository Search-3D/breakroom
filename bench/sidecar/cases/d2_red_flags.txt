+package cart
+
+import (
+	"database/sql"
+	"time"
+)
+
+const schema = `CREATE TABLE IF NOT EXISTS cart_items (
+	sid        TEXT NOT NULL,
+	product_id TEXT NOT NULL,
+	qty        INTEGER NOT NULL,
+	created_at TIMESTAMP NOT NULL,
+	PRIMARY KEY (sid, product_id)
+)`
+
+// SQLStore persists carts in the shared SQLite database.
+type SQLStore struct{ db *sql.DB }
+
+func NewSQLStore(db *sql.DB) *SQLStore {
+	db.Exec(schema)
+	return &SQLStore{db: db}
+}
+
+func (s *SQLStore) Add(sid string, it Item) {
+	s.db.Exec(`INSERT INTO cart_items (sid, product_id, qty, created_at)
+		VALUES (?, ?, ?, ?)
+		ON CONFLICT (sid, product_id) DO UPDATE SET qty = qty + excluded.qty`,
+		sid, it.ProductID, it.Qty, time.Now())
+}
+
+func (s *SQLStore) Get(sid string) []Item {
+	rows, err := s.db.Query(`SELECT product_id, qty FROM cart_items WHERE sid = ? ORDER BY created_at`, sid)
+	if err != nil {
+		return nil
+	}
+	defer rows.Close()
+	out := []Item{}
+	for rows.Next() {
+		var it Item
+		rows.Scan(&it.ProductID, &it.Qty)
+		out = append(out, it)
+	}
+	return out
+}
