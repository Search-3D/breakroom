+// stop lists terms dropped at index and query time.
+var stop = map[string]bool{"a": true, "an": true, "the": true, "of": true, "and": true}
+
+// Tokenize splits text on Unicode word boundaries, NFC-normalizes and
+// lowercases each word, and drops stop words and pure punctuation.
+func Tokenize(text string) []string {
+	seg := segment.NewWordSegmenter(strings.NewReader(norm.NFC.String(text)))
+	var out []string
+	for seg.Segment() {
+		if seg.Type() == segment.None {
+			continue
+		}
+		w := strings.ToLower(seg.Text())
+		if stop[w] || !hasLetterOrDigit(w) {
+			continue
+		}
+		out = append(out, w)
+	}
+	return out
+}
